Buttons

Primary Button

<!-- Button HTML -->
<button class="nidi-button--primary">Default</button>

<!-- Button CSS -->
button {
    padding: 12px 24px;
    font-size: 16px;
    font-weight: 600;
    font-family: 'Quicksand', sans-serif;
    border: 2px solid transparent;
    border-radius: 24px;
    cursor: pointer;
    transition: all 0.15s;
}

/* Default State */
.nidi-button--primary {
    background: #6A0011;
    color: #FFFFFF;
}

/* Hover State */
.nidi-button--primary:hover {
    background: #6A0011;
    border: 2px solid #370009;
    color: #FFFFFF;
}

/* Pressed State */
.nidi-button--primary:active {
    background: #370009;
    border: 2px solid transparent;
    color: #FFFFFF;
}

/* Disabled State */
.nidi-button--primary:disabled {
    background: #b8b8b8;
    color: #5A4245;
    border: 2px solid transparent;
}
